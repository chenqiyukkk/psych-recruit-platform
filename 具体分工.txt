from datetime import datetime, timedelta
import hashlib

fake = Faker('zh_CN')

def generate_users(count=100):
    """生成测试用户"""
    users = []
    for i in range(count):
        role = random.choice(['SUBJECT', 'RESEARCHER', 'ADMIN'])
        users.append({
            'username': f'user{i}',
            'password': hashlib.sha256('123456'.encode()).hexdigest(),
            'phone': fake.phone_number(),
            'email': fake.email(),
            'role': role,
            'reputation_score': 100 if role == 'SUBJECT' else None,
            'researcher_rating': 5.0 if role == 'RESEARCHER' else None
        })
    return users

def generate_participant_profiles(user_ids, count=80):
    """生成被试档案"""
    profiles = []
    genders = ['MALE', 'FEMALE', 'OTHER']
    majors = ['心理学类', '计算机类', '文学类', '理学类']
    handedness = ['LEFT', 'RIGHT', 'MIXED']
    
    for i in range(count):
        user_id = random.choice(user_ids)
        profiles.append({
            'user_id': user_id,
            'anonymous_id': f'P-2025-{str(i+1).zfill(3)}',
            'age_group': random.choice(['18-22', '23-25', '26-30']),
            'gender': random.choice(genders),
            'major_category': random.choice(majors),
            'handedness': random.choice(handedness),
            'reputation_score': random.randint(60, 100)
        })
    return profiles

def generate_experiments(researcher_ids, count=50):
    """生成实验项目"""
    experiments = []
    statuses = ['DRAFT', 'PUBLISHED', 'RECRUITING', 'FULL', 'ONGOING', 'COMPLETED']
    risk_levels = ['LOW', 'MEDIUM', 'HIGH']
    
    for i in range(count):
        start_time = fake.date_time_between(start_date='+1d', end_date='+30d')
        end_time = start_time + timedelta(hours=random.randint(2, 8))
        
        experiments.append({
            'title': fake.sentence(nb_words=6),
            'description': fake.paragraph(),
            'location': fake.address(),
            'start_time': start_time,
            'end_time': end_time,
            'ethics_approval_no': f'IRB-2025-PSY-{str(i+1).zfill(3)}',
            'risk_level': random.choice(risk_levels),
            'payment_amount': round(random.uniform(50, 200), 2),
            'payment_method': 'OFFLINE',
            'payment_description': '实验结束后现场支付',
            'screening_criteria': '{"include": {"age_range": [18, 25], "gender": "female"}}',
            'exclude_tags': '["fMRI", "情绪类"]',
            'status': random.choice(statuses),
            'organizer_id': random.choice(researcher_ids)
        })
    return experiments

def insert_to_database(users, profiles, experiments):
    """插入数据到MySQL"""
    conn = mysql.connector.connect(
        host='localhost',
        user='root',
        password='password',
        database='psychology_platform'
    )
    cursor = conn.cursor()
    
    # 插入用户
    user_ids = []
    for user in users:
        cursor.execute("""
            INSERT INTO users (username, password, phone, email, role, reputation_score)
            VALUES (%s, %s, %s, %s, %s, %s)
        """, (user['username'], user['password'], user['phone'], 
              user['email'], user['role'], user.get('reputation_score')))
        user_ids.append(cursor.lastrowid)
    
    # 插入被试档案
    for profile in profiles:
        cursor.execute("""
            INSERT INTO participant_profiles (user_id, anonymous_id, age_group, gender, 
                                             major_category, handedness, reputation_score)
            VALUES (%s, %s, %s, %s, %s, %s, %s)
        """, (profile['user_id'], profile['anonymous_id'], profile['age_group'],
              profile['gender'], profile['major_category'], profile['handedness'],
              profile['reputation_score']))
    
    # 插入实验
    for experiment in experiments:
        cursor.execute("""
            INSERT INTO experiments (title, description, location, start_time, end_time,
                                    ethics_approval_no, risk_level, payment_amount, 
                                    payment_method, payment_description, screening_criteria,
                                    exclude_tags, status, organizer_id)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
        """, (experiment['title'], experiment['description'], experiment['location'],
              experiment['start_time'], experiment['end_time'], experiment['ethics_approval_no'],
              experiment['risk_level'], experiment['payment_amount'], experiment['payment_method'],
              experiment['payment_description'], experiment['screening_criteria'],
              experiment['exclude_tags'], experiment['status'], experiment['organizer_id']))
    
    conn.commit()
    cursor.close()
    conn.close()
    
    print(f"成功生成：")
    print(f"  - {len(users)} 个用户")
    print(f"  - {len(profiles)} 个被试档案")
    print(f"  - {len(experiments)} 个实验项目")

if __name__ == '__main__':
    users = generate_users(100)
    # 分离研究者ID
    researcher_ids = [i+1 for i, user in enumerate(users) if user['role'] == 'RESEARCHER']
    # 生成被试档案（80个）
    profiles = generate_participant_profiles(list(range(1, 101)), 80)
    # 生成实验（50个）
    experiments = generate_experiments(researcher_ids, 50)
    # 插入数据库
    insert_to_database(users, profiles, experiments)
```

```python
import os
import zipfile

def create_zip():
    folder_name = "20210001-张三-心理学被试招募平台"
    zip_name = "20210001-张三-心理学被试招募平台.zip"
    
    with zipfile.ZipFile(zip_name, 'w', zipfile.ZIP_DEFLATED) as zipf:
        # 添加文档
        docs = [
            "20210001-张三-需求分析报告.pdf",
            "20210001-张三-系统建模报告.pdf",
            # ... 其他文档
        ]
        for doc in docs:
            zipf.write(doc, arcname=os.path.join(folder_name, doc))
        
        # 添加源代码
        for root, dirs, files in os.walk('backend'):
            for file in files:
                if not file.endswith(('.class', '.jar')):
                    file_path = os.path.join(root, file)
                    zipf.write(file_path, arcname=os.path.join(folder_name, file_path))
        
        # 添加README
        zipf.write('README.md', arcname=os.path.join(folder_name, 'README.md'))
    
    print(f"打包完成：{zip_name}")

if __name__ == '__main__':
    create_zip()
